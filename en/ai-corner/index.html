# Automated catalog photography generation (Python + ComfyUI API)
import requests, json, pathlib

def generate_catalog_batch(products_dir: str, lora_model: str):
    api_url = "http://127.0.0.1:8188/prompt"
    for raw_photo in pathlib.Path(products_dir).glob("*.png"):
        payload = {
            "model": "flux1-dev-fp8.safetensors",
            "lora": lora_model,
            "controlnet_depth": str(raw_photo),
            "lighting": "IC-Light Studio Softbox 5500K",
            "upscale": 4096
        }
        res = requests.post(api_url, json={"prompt": payload})
        print(f"[OK] 4K catalog frame generated for: {raw_photo.name}")

generate_catalog_batch("./raw_products", "client_brand_v3.safetensors")
// Node.js SDK: Automated video motion rendering for e-commerce products
import { RunwayML } from '@runwayml/sdk';
import fs from 'node:fs/promises';

const client = new RunwayML({ apiKey: process.env.RUNWAY_SECRET });

export async function renderProductMotion(productImageUrl, sku) {
  const task = await client.imageToVideo.create({
    model: 'gen3a_turbo',
    promptImage: productImageUrl,
    promptText: 'Slow cinematic orbit camera around product, golden hour reflections, 60fps',
  });
  console.log(`[AI VIDEO] Rendering ad spot for SKU: ${sku} | Task: ${task.id}`);
  return task;
}
